# The pinned ev3dev Stretch cross image provides CMake 3.7.2.
cmake_minimum_required(VERSION 3.7)
project(phantom_ev3 LANGUAGES CXX)
# Export facts about the selected compiler; runners do not choose it.
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" --version
    OUTPUT_VARIABLE compiler_version RESULT_VARIABLE compiler_version_result)
if(NOT compiler_version_result EQUAL 0)
    message(FATAL_ERROR "Cannot read selected compiler version")
endif()
string(REGEX MATCH "^[^\r\n]+" compiler_version_line "${compiler_version}")
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_sha256)
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpmachine
    OUTPUT_VARIABLE compiler_target OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE compiler_target_result)
if(NOT compiler_target_result EQUAL 0 OR compiler_target STREQUAL "")
    message(FATAL_ERROR "Cannot read selected compiler target")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/toolchain-facts.txt"
    "${compiler_version_line}\n${compiler_sha256}  ${CMAKE_CXX_COMPILER}\ncompiler_target=${compiler_target}\n")
option(PHANTOM_DIAGNOSTIC "Enable diagnostic surfaces only" OFF)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(phantom-core src/controller_core.cpp src/mock_motor_backend.cpp src/ev3_sysfs_backend.cpp src/runtime_guard.cpp src/runtime_profile.cpp src/robot_data.cpp src/network_ipc.cpp src/json_reader.cpp src/event_log.cpp)
target_include_directories(phantom-core PUBLIC include ../protocol/generated)
target_compile_options(phantom-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(phantom-core PUBLIC PHANTOM_BUILD_DIAGNOSTIC=$<BOOL:${PHANTOM_DIAGNOSTIC}>)
add_executable(phantom-controller src/controller_main.cpp)
target_link_libraries(phantom-controller PRIVATE phantom-core)
add_executable(phantom-safety safety/main.cpp)
target_link_libraries(phantom-safety PRIVATE phantom-core)
add_library(phantom-network-core src/network_wire.cpp src/network_server.cpp src/controller_ipc_client.cpp)
target_link_libraries(phantom-network-core PUBLIC phantom-core ${CMAKE_DL_LIBS})
target_compile_options(phantom-network-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(phantom-network src/network_main.cpp)
target_link_libraries(phantom-network PRIVATE phantom-network-core)
target_compile_options(phantom-network PRIVATE -Wall -Wextra -Wpedantic)
add_executable(phantom-mdns src/mdns_main.cpp)
target_link_libraries(phantom-mdns PRIVATE phantom-network-core)
target_compile_options(phantom-mdns PRIVATE -Wall -Wextra -Wpedantic)
option(PHANTOM_BUILD_TESTS "Build host test suites and fixtures" ON)
add_library(phantom-maintenance-core src/maintenance_support.cpp src/device_integration.cpp)
target_link_libraries(phantom-maintenance-core PUBLIC phantom-network-core)
target_compile_options(phantom-maintenance-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(phantom-maintenance src/maintenance_main.cpp)
target_link_libraries(phantom-maintenance PRIVATE phantom-maintenance-core)
target_compile_options(phantom-maintenance PRIVATE -Wall -Wextra -Wpedantic)
add_executable(phantom-device-integration-test src/device_integration_main.cpp)
target_link_libraries(phantom-device-integration-test PRIVATE phantom-maintenance-core)
if(PHANTOM_BUILD_TESTS)
enable_testing()
find_package(Threads REQUIRED)
add_executable(phantom-maintenance-tests tests/maintenance_tests.cpp)
target_link_libraries(phantom-maintenance-tests PRIVATE phantom-maintenance-core)
target_compile_options(phantom-maintenance-tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME phantom-maintenance-tests COMMAND phantom-maintenance-tests)
add_executable(phantom-network-server-tests tests/network_server_tests.cpp)
target_link_libraries(phantom-network-server-tests PRIVATE phantom-network-core Threads::Threads)
target_compile_definitions(phantom-network-server-tests PRIVATE PHANTOM_REPO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME phantom-network-server-tests COMMAND phantom-network-server-tests)
add_library(phantom-fake-avahi-common SHARED tests/fake_avahi.cpp)
set_target_properties(phantom-fake-avahi-common PROPERTIES OUTPUT_NAME avahi-common SOVERSION 3 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-libraries")
add_library(phantom-fake-avahi-client SHARED tests/fake_avahi.cpp)
target_compile_definitions(phantom-fake-avahi-client PRIVATE PHANTOM_FAKE_AVAHI_CLIENT=1)
target_link_libraries(phantom-fake-avahi-client PRIVATE phantom-fake-avahi-common)
set_target_properties(phantom-fake-avahi-client PROPERTIES OUTPUT_NAME avahi-client SOVERSION 3 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-libraries")
add_executable(phantom-mdns-tests tests/mdns_tests.cpp)
target_link_libraries(phantom-mdns-tests PRIVATE phantom-network-core)
add_dependencies(phantom-mdns-tests phantom-fake-avahi-client)
add_test(NAME phantom-mdns-tests COMMAND phantom-mdns-tests)
set_tests_properties(phantom-mdns-tests PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/test-libraries")
add_executable(phantom-core-tests tests/core_tests.cpp)
target_link_libraries(phantom-core-tests PRIVATE phantom-core)
add_test(NAME phantom-core-tests COMMAND phantom-core-tests)
add_executable(phantom-network-ipc-tests tests/network_ipc_tests.cpp)
target_link_libraries(phantom-network-ipc-tests PRIVATE phantom-core)
add_test(NAME phantom-network-ipc-tests COMMAND phantom-network-ipc-tests)
add_executable(phantom-config-tests tests/config_tests.cpp)
target_link_libraries(phantom-config-tests PRIVATE phantom-core)
target_compile_definitions(phantom-config-tests PRIVATE PHANTOM_REPO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME phantom-config-tests COMMAND phantom-config-tests)
add_executable(phantom-ipc-vector-tests tests/ipc_vector_tests.cpp)
target_link_libraries(phantom-ipc-vector-tests PRIVATE phantom-core)
target_compile_definitions(phantom-ipc-vector-tests PRIVATE PHANTOM_REPO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME phantom-ipc-vector-tests COMMAND phantom-ipc-vector-tests)
add_executable(phantom-quest-interop-fixture tests/support/quest_interop_fixture.cpp)
target_link_libraries(phantom-quest-interop-fixture PRIVATE phantom-core)
target_compile_options(phantom-quest-interop-fixture PRIVATE -Wall -Wextra -Wpedantic)
add_dependencies(phantom-quest-interop-fixture phantom-network)
endif()
install(TARGETS phantom-controller phantom-safety phantom-network phantom-mdns phantom-maintenance phantom-device-integration-test RUNTIME DESTINATION bin)
