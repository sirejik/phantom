# The pinned ev3dev Stretch cross image provides CMake 3.7.2.
cmake_minimum_required(VERSION 3.7)
project(phantom_ev3 LANGUAGES CXX)
option(PHANTOM_DIAGNOSTIC "Enable diagnostic surfaces only" OFF)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(phantom-core src/controller_core.cpp src/mock_motor_backend.cpp src/ev3_sysfs_backend.cpp src/runtime_guard.cpp src/runtime_profile.cpp src/network_ipc.cpp src/json_reader.cpp src/event_log.cpp)
target_include_directories(phantom-core PUBLIC include ../protocol/generated)
target_compile_options(phantom-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(phantom-core PUBLIC PHANTOM_BUILD_DIAGNOSTIC=$<BOOL:${PHANTOM_DIAGNOSTIC}>)
add_executable(phantom-controller src/controller_main.cpp)
target_link_libraries(phantom-controller PRIVATE phantom-core)
add_executable(phantom-safety safety/main.cpp)
target_link_libraries(phantom-safety PRIVATE phantom-core)
enable_testing()
add_executable(phantom-core-tests tests/core_tests.cpp)
target_link_libraries(phantom-core-tests PRIVATE phantom-core)
add_test(NAME phantom-core-tests COMMAND phantom-core-tests)
add_executable(phantom-network-ipc-tests tests/network_ipc_tests.cpp)
target_link_libraries(phantom-network-ipc-tests PRIVATE phantom-core)
add_test(NAME phantom-network-ipc-tests COMMAND phantom-network-ipc-tests)
add_executable(phantom-config-tests tests/config_tests.cpp)
target_link_libraries(phantom-config-tests PRIVATE phantom-core)
target_compile_definitions(phantom-config-tests PRIVATE PHANTOM_REPO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME phantom-config-tests COMMAND phantom-config-tests)
add_executable(phantom-ipc-vector-tests tests/ipc_vector_tests.cpp)
target_link_libraries(phantom-ipc-vector-tests PRIVATE phantom-core)
target_compile_definitions(phantom-ipc-vector-tests PRIVATE PHANTOM_REPO_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME phantom-ipc-vector-tests COMMAND phantom-ipc-vector-tests)
install(TARGETS phantom-controller phantom-safety RUNTIME DESTINATION bin)
